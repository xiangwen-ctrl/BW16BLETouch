cmake_minimum_required(VERSION 3.15)
project(BW16BLETouchHostTests LANGUAGES CXX)
enable_testing()
foreach(test reconnect paths)
    add_executable(${test} ${test}.cpp ../src/BW16BLETouch.cpp)
    target_include_directories(${test} PRIVATE fakes ../src)
    target_compile_features(${test} PRIVATE cxx_std_11)
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
